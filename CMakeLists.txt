cmake_minimum_required(VERSION 3.16)
project(raymotion VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
find_package(Threads REQUIRED)
add_library(raymotion_engine STATIC
  src/engine/scene.cpp src/engine/obj.cpp src/engine/bvh.cpp
  src/engine/pathtrace.cpp src/engine/renderer.cpp)
option(RAYMOTION_METAL "Enable Metal GPU rendering on macOS" ON)
if(APPLE AND RAYMOTION_METAL)
  enable_language(OBJCXX)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS src/engine/pathtrace.metal src/engine/metal_types.h)
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/engine/pathtrace.metal" METAL_SOURCE)
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/engine/metal_types.h" METAL_TYPES)
  string(REPLACE "#pragma once" "" METAL_TYPES "${METAL_TYPES}")
  string(REPLACE "#include \"metal_types.h\"" "${METAL_TYPES}" METAL_SOURCE "${METAL_SOURCE}")
  configure_file(src/engine/metal_source.h.in metal_source.h @ONLY)
  target_sources(raymotion_engine PRIVATE src/engine/metal.mm)
  set_source_files_properties(src/engine/metal.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
  set_target_properties(raymotion_engine PROPERTIES OBJCXX_STANDARD 17 OBJCXX_STANDARD_REQUIRED YES)
  target_include_directories(raymotion_engine PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  target_link_libraries(raymotion_engine PUBLIC "-framework Foundation" "-framework Metal")
else()
  target_sources(raymotion_engine PRIVATE src/engine/metal_stub.cpp)
endif()
target_compile_features(raymotion_engine PUBLIC cxx_std_17)
target_include_directories(raymotion_engine PUBLIC include)
target_link_libraries(raymotion_engine PUBLIC Threads::Threads)
if(NOT MSVC)
  target_compile_options(raymotion_engine PRIVATE -Wall -Wextra)
endif()
install(TARGETS raymotion_engine ARCHIVE DESTINATION lib)
install(DIRECTORY include/raymotion DESTINATION include)
install(PROGRAMS src/cli/raymotion DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_cli.py)
  add_executable(test_bvh tests/test_bvh.cpp)
  target_link_libraries(test_bvh PRIVATE raymotion_engine)
  add_test(NAME bvh COMMAND test_bvh)
  add_executable(test_metal tests/test_metal.cpp)
  target_link_libraries(test_metal PRIVATE raymotion_engine)
  add_test(NAME metal COMMAND test_metal)
  set_tests_properties(metal PROPERTIES SKIP_RETURN_CODE 77)
endif()
