cmake_minimum_required(VERSION 3.16)
project(raymotion VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
find_package(Threads REQUIRED)
add_library(raymotion_engine STATIC
  src/engine/scene.cpp src/engine/obj.cpp src/engine/bvh.cpp
  src/engine/pathtrace.cpp src/engine/renderer.cpp)
target_compile_features(raymotion_engine PUBLIC cxx_std_17)
target_include_directories(raymotion_engine PUBLIC include)
target_link_libraries(raymotion_engine PUBLIC Threads::Threads)
if(NOT MSVC)
  target_compile_options(raymotion_engine PRIVATE -Wall -Wextra)
endif()
install(TARGETS raymotion_engine ARCHIVE DESTINATION lib)
install(DIRECTORY include/raymotion DESTINATION include)
install(PROGRAMS src/cli/raymotion DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_cli.py)
  add_executable(test_bvh tests/test_bvh.cpp)
  target_link_libraries(test_bvh PRIVATE raymotion_engine)
  add_test(NAME bvh COMMAND test_bvh)
endif()
